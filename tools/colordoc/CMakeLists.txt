cmake_minimum_required(VERSION 4.2.0)
project(colordoc)

find_package(mupdf REQUIRED)

set(CMAKE_CXX_STANDARD 20)

set(PDF_CLASSIFIER_LIB_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../pdf_classifier_lib pdf_classifier_lib_build) # ! temp

# pdf_classifier_lib fetches nlohmann_json, so it can only be found after the
# add_subdirectory above. Its source dir is set in the lib's scope, not this one.
find_package(nlohmann_json 3.12.0 REQUIRED HINTS "${CMAKE_BINARY_DIR}/_deps/nlohmann_json-build")
include(FetchContent)
FetchContent_GetProperties(nlohmann_json)

# The target name is fixed: pdf_classifier_core/build.rs links
# `classifier_intermediary` out of ${CLASSIFIER_BUILD_DIR}/lib.
add_library(classifier_intermediary STATIC classes.cpp color_page.cpp)

target_link_libraries(classifier_intermediary PRIVATE mupdf::mupdf PUBLIC pdf_classifier_lib nlohmann_json::nlohmann_json)

target_include_directories(classifier_intermediary PRIVATE
    ${CLASSIFIER_INCLUDE_DIR} # provided during build via the Python Builder
    ${CMAKE_INSTALL_PREFIX}/include
)

target_sources(classifier_intermediary
    PUBLIC
    FILE_SET HEADERS
    BASE_DIRS ${CMAKE_CURRENT_SOURCE_DIR}
    FILES
    palette.hpp # generated by palette.py; run main.py (or palette.py) first
    color_page.hpp
    classes.hpp
)

install(TARGETS classifier_intermediary
    FILE_SET HEADERS DESTINATION include/generated
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
)

install(DIRECTORY ${nlohmann_json_SOURCE_DIR}/include/
    DESTINATION include
)

enable_testing()

add_subdirectory(tests)
